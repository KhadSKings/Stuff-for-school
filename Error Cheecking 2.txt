def valid_num(question, low, high):
    error = f"Buddy, that is not a integer between {low} and {high}."
    while True:
        try:
            response = int(input(question))
            #If the response >= to the low nr or <= to high nr
            if low <= response >= high:
                break #It stops the loop

            else:
                print(f"{error}/n")
        except ValueError:
            print(f"{error}")
        return response #This makes the response avalibe to be used outside the function
